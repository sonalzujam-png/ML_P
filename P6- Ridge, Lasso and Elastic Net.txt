import matplotlib.pyplot as plt
from sklearn.linear_model import Ridge, Lasso, ElasticNet

X = [[1], [2], [3], [4], [5]]
y = [10, 20, 30, 40, 50]

m1 = Ridge()
m2 = Lasso()
m3 = ElasticNet()

m1.fit(X, y)
m2.fit(X, y)
m3.fit(X, y)

plt.plot(y, label="Actual")
plt.plot(m1.predict(X), label="Ridge")
plt.plot(m2.predict(X), label="Lasso")
plt.plot(m3.predict(X), label="ElasticNet")

plt.legend()
plt.show()
